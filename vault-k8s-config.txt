################################## CREAR POLITICA Y TOKEN ########################################################
kubectl exec -i -n vault vault-0 -- vault policy write busa-login-manager-policy - <<EOF
# Aquí va el contenido HCL de tu política
path "secret/data/busa-login-manager" {
  capabilities = ["read","update"]
}
path "secret/metadata/busa-login-manager" {
  capabilities = ["read", "list"]
}
path "sys/capabilities-self" {
  capabilities = ["update"]
}
EOF
#
kubectl exec -n vault vault-0 -- vault token create -policy="busa-login-manager-policy"

# crear el archivo test-vault-auth.yaml
apiVersion: v1
kind: Pod
metadata:
  name: test-vault-auth
spec:
  serviceAccountName: mi-app-sa
  restartPolicy: Never
  containers:
  - name: test
    image: hashicorp/vault:latest
    env:
    - name: VAULT_ADDR
      value: "http://vault.vault.svc.cluster.local:8200"
    command: ["sh", "-c"]
    args:
    - |
      # Leer el token del volumen proyectado (NO el de la ruta por defecto)
      JWT=$(cat /var/run/secrets/tokens/vault-token)
      # Autenticarse en Vault
      VAULT_TOKEN=$(vault write -field=token auth/kubernetes/login role=mi-app-role jwt=$JWT)
      # Leer el secreto
      VAULT_TOKEN=$VAULT_TOKEN vault kv get secret/app1
    volumeMounts:
    - name: vault-token
      mountPath: /var/run/secrets/tokens
      readOnly: true
  volumes:
  - name: vault-token
    projected:
      sources:
      - serviceAccountToken:
          audience: vault
          expirationSeconds: 3600
          path: vault-token

kubectl apply -f test-vault-auth.yaml
kubectl logs test-vault-auth
